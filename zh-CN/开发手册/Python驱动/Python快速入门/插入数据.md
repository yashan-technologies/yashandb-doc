使用execute()方法执行INSERT语句插入数据。

## 插入单条数据

```python
cursor = connection.cursor()

# 插入单条数据
cursor.execute("INSERT INTO example_table VALUES (1, 'test1')")

# 提交事务
connection.commit()

cursor.close()
```

## 批量插入数据

```python
cursor = connection.cursor()

# 批量插入数据（使用位置绑定）
data = [
    (2, 'test2'),
    (3, 'test3'),
    (4, 'test4')
]
cursor.executemany("INSERT INTO example_table VALUES (:1, :2)", data)

# 提交事务
connection.commit()

cursor.close()
```

## 插入带参数的数据

```python
cursor = connection.cursor()

# 使用参数化插入（使用位置绑定）
cursor.execute("INSERT INTO example_table VALUES (:1, :2)", (5, 'test5'))

# 提交事务
connection.commit()

cursor.close()
```

## 完整示例

```python
import yaspy

try:
    # 连接数据库
    conn = yaspy.connect(dsn='sales/sales@192.168.1.2:1688')
    cursor = conn.cursor()

    # 创建测试表
    cursor.execute('DROP TABLE IF EXISTS products')
    cursor.execute('''
        CREATE TABLE products (
            id INT PRIMARY KEY,
            name VARCHAR(100),
            price DECIMAL(10, 2),
            stock INT
        )
    ''')
    conn.commit()
    print('Table created')

    # 插入单条数据
    cursor.execute("INSERT INTO products VALUES (1, 'Laptop', 5999.00, 50)")
    conn.commit()
    print('Single record inserted')

    # 批量插入数据（使用命名绑定）
    products = [
        {'id': 2, 'name': 'Mouse', 'price': 99.00, 'stock': 200},
        {'id': 3, 'name': 'Keyboard', 'price': 299.00, 'stock': 150},
        {'id': 4, 'name': 'Monitor', 'price': 1999.00, 'stock': 80}
    ]
    cursor.executemany(
        "INSERT INTO products VALUES (:id, :name, :price, :stock)",
        products
    )
    conn.commit()
    print('Batch records inserted')

    # 使用参数化插入（使用命名绑定，换用新游标避免绑定冲突）
    cursor2 = conn.cursor()
    cursor2.execute(
        "INSERT INTO products VALUES (:id, :name, :price, :stock)",
        {'id': 5, 'name': 'Headphones', 'price': 599.00, 'stock': 100}
    )
    conn.commit()
    cursor2.close()
    print('Parameterized record inserted')

    # 查询验证所有数据
    cursor.execute('SELECT * FROM products ORDER BY id')
    rows = cursor.fetchall()
    print(f'Total products: {len(rows)}')
    for row in rows:
        print(f'  ID: {row[0]}, Name: {row[1]}, Price: {row[2]}, Stock: {row[3]}')

    # 清理测试表
    cursor.execute('DROP TABLE products')
    conn.commit()
    print('Test table cleaned up')

    cursor.close()
    conn.close()
    print('Connection closed')

except Exception as e:
    print(f'Error: {e}')
```
