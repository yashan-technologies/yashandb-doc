The server configuration file lists the physical information of all instances of the YashanDB server.

### Typical Example: Standalone Deployment

***Example***: for hosts.toml

```toml
# The following content is based on the actual display results
uuid = "64f04f0e999c72db72bd565762e6a4ea"   # Automatically generated by the system, not recommended to modify
cluster = "yashandb"    # Cluster name. Changes made after installation will not take effect unless reinstalled
yas_type = "SE"   # Deployment type
secret_key = "e8540d7ef94a2c43" # Automatically generated by the system, not recommended to modify
isLocal = true # Indicates whether it is a local installation. If it is local, no SSH username and password need to be specified
add_yasdba = true # Indicates whether to add the current user to the YASDBA user group
plugins = ["all"] # Installed plugins
[om]	# yasom information
  hostid = "host0001"	# Identifier for the main yasom's server
  [om.config]	
    LISTEN_ADDR = "127.0.0.1:1675"   # yasom listening port

[[host]]
  hostid = "host0001"   # Identifier for the server where the database instance is located
  user = "yashan"       # SSH user for logging into the server (database installation user)
  password = "password" # SSH login password
  ip = "192.168.1.2"    # Database server
  port = 22	        # SSH port for logging into the server
  path = "/data/yashan/yasdb_home"	    # Product installation path, ensure this path is empty; changes made after installation will not take effect unless reinstalled
  log_path = "/data/yashan/log"         # yasom and yasagent logs
  [host.yasagent]
    [host.yasagent.config]
      LISTEN_ADDR = "192.168.1.2:1676"	# yasagent listening port
```
