The standalone configuration file lists the relevant information for the YashanDB server-side database and the logical cluster architecture.

Among them, the parameter-related configuration information is divided into the following two parts:

- [Database Creation Parameters](../Database Creation Parameters): These parameters control the relevant information when creating a database instance and are configured at the [[group]] instance group level, ensuring that multiple instances within each instance group create consistent databases.
- [Configuration Parameters](../../../Reference Manual/Configuration Parameters): The configuration parameter information for each database instance is configured at the [[group.node]] instance level.



>**Caution**:
>
> The sample content of the configuration files provided in this chapter is mainly used to introduce the composition and meaning of configuration items. To avoid introducing unnecessary errors, it is **not recommended** to directly copy the sample content to manually construct configuration files.



***Example***: for yashandb.toml
```toml
# The following content is subject to the actual display results
cluster = "yashandb" # Modifications after installation will not take effect unless reinstalled
create_simple_schema = false # If true, example SQL will execute after deployment
sys_password = "encrypt_sys_password" # The encrypted password for the super administrator SYS user, manual changes are not supported; if a password needs to be re-specified, please do so with the --sys-password parameter during deployment
uuid = "64f04f0e999c72db72bd565762e6a4ea" # Automatically generated by the system, modification is not recommended
mode = "YASHAN" # Syntax mode, only applicable to Standalone Deployment 
yas_type = "SE" # Deployment mode, modifications after installation will not take effect unless reinstalled

[[group]]
  group_type = "db" # Modifications after installation will not take effect unless reinstalled
  name = "dbg1" # Modifications after installation will not take effect unless reinstalled
  [group.config] # All database creation parameters can be configured; default values are used if not configured
    CHARACTER_SET = "utf8"
    ISARCHIVELOG = true
    REDO_FILE_NUM = 4
    REDO_FILE_SIZE = "128M"

  [[group.node]]
    data_path = "/data/yashan/yasdb_data" # This is the DATA directory; modifications after installation will not take effect unless reinstalled
    hostid = "host0001" # Modifications after installation will not take effect unless reinstalled
    role = 1    # Database primary/standby role; modifications after installation will not take effect unless reinstalled
    [group.node.config] # All database parameters can be configured; default values are used if not configured, and modifications after installation can take effect (read-only parameters cannot be modified)
      LISTEN_ADDR = "192.168.1.2:1688"
      REPLICATION_ADDR = "192.168.1.2:1689"
      RUN_LOG_FILE_PATH = "/data/yashan/log"
      RUN_LOG_LEVEL = "INFO"
      SLOW_LOG_FILE_PATH = "/data/yashan/log"
    [group.node.mysql_config] # Configurable global variables in mysql mode that support file modifications

  # In high availability mode, the following standby database parameter information will be added.
  [[group.node]]
    data_path = "/data/yashan/yasdb_data"
    hostid = "host0002"
    role = 2
    [group.node.config]
      LISTEN_ADDR = "192.168.1.3:1688"
      REPLICATION_ADDR = "192.168.1.3:1690"
      RUN_LOG_FILE_PATH = "/data/yashan/log"
      RUN_LOG_LEVEL = "INFO"
      SLOW_LOG_FILE_PATH = "/data/yashan/log"
    [group.node.mysql_config]

  [[group.node]]
    data_path = "/data/yashan/yasdb_data"
    hostid = "host0003"
    role = 2
    [group.node.config]
      LISTEN_ADDR = "192.168.1.4:1688"
      REPLICATION_ADDR = "192.168.1.4:1690"
      RUN_LOG_FILE_PATH = "/data/yashan/log"
      RUN_LOG_LEVEL = "INFO"
      SLOW_LOG_FILE_PATH =  "/data/yashan/log"
    [group.node.mysql_config]

```
