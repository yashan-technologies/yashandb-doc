## Description

`java.sql.Connection` represents that it is the core interface in the JDBC specification that represents the physical connection between a Java application and a database, and serves as the fundamental entry point for all database operations. Whether executing SQL statements, calling stored procedures, or managing transactions, a connection to the database must first be established via `Connection` and obtained through the DriverManager.getConnection(url, user, password) method.

Detailed interface specifications, please refer to [The JDBC Official Website](https://docs.oracle.com/en/java/javase/17/docs/api/java.sql/java/sql/Connection.html).

## Example

```java
package quickstart;

import java.sql.Connection;
import java.sql.DatabaseMetaData;
import java.sql.DriverManager;
import java.sql.Savepoint;
import java.sql.SQLException;
import java.util.Properties;

public class ConnectionManager {

    public static Connection getConnection() throws SQLException {
        String url = "jdbc:yasdb://192.168.1.2:1688/yasdb";
        Properties info = new Properties();
        info.setProperty("user", "sales");
        info.setProperty("password", "sales");
        Class.forName("com.yashandb.jdbc.Driver");
        return DriverManager.getConnection(url, info);
    }

    public void completeConnectionDemo() throws SQLException {
        try (Connection conn = getConnection()) {
            // 1. Set connection properties
            conn.setAutoCommit(false);
            conn.setReadOnly(false);
            conn.setTransactionIsolation(Connection.TRANSACTION_READ_COMMITTED);

            // 2. Get metadata
            DatabaseMetaData metaData = conn.getMetaData();
            System.out.println("Connected to: " + metaData.getDatabaseProductName());

            // 3. Execute transaction
            executeTransaction(conn);

            // 4. Check connection status
            if (conn.isValid(10)) {
                System.out.println("Connection is valid");
            }
        }
    }

    private void executeTransaction(Connection conn) throws SQLException {
        Savepoint savepoint = null;
        try {
            // Create savepoint
            savepoint = conn.setSavepoint("BEFORE_OPERATIONS");

            // Execute multiple operations
            executeOperations(conn);

            // Commit transaction
            conn.commit();
            System.out.println("Transaction committed successfully");

        } catch (SQLException e) {
            if (savepoint != null) {
                // Rollback to savepoint
                conn.rollback(savepoint);
                System.out.println("Rolled back to savepoint");
            } else {
                // Full rollback
                conn.rollback();
                System.out.println("Transaction rolled back");
            }
            throw e;
        }
    }

    private void executeOperations(Connection conn) throws SQLException {
        // Execute specific operations
    }
}
```
